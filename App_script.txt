Um ponto que chamou minha atenção
Vocês possuem duas funções:

doGet()

getOperadoresPorTL()

Mas apenas doGet() pode ser chamada diretamente pelo site via URL do Web App.

A função getOperadoresPorTL() não fica acessível pela URL da forma como está escrita. Ela só pode ser chamada por outro código dentro do próprio Apps Script.

Isso não significa que esteja errado, apenas é um ponto que vou verificar quando analisar o JavaScript do projeto. Dependendo de como ele faz a requisição, isso pode ser justamente a origem do problema.

    let escalaA = Array(18).fill(null);
    let escalaB = Array(18).fill(null);
    let nomeTL = "";

    function classeProdutividade(val) {
        if (val === undefined || val === null || val === "") return "bolinha-amarela";
        let num = parseFloat(val);
        if (isNaN(num)) return "bolinha-amarela";
        
        if (num >= 260) return "bolinha-verde";
        if (num >= 200) return "bolinha-amarela";
        return "bolinha-vermelha"; 
    }

    function classeOciosidade(val) {
        if (val === undefined || val === null || val === "") return "bolinha-amarela";
        let num = parseFloat(val);
        if (isNaN(num)) return "bolinha-amarela";
        
        if (num <= 10) return "bolinha-verde";
        if (num >= 12) return "bolinha-vermelha";
        return "bolinha-amarela";
    }

    function render() {
        document.getElementById('nome-tl-display').innerText = nomeTL || "---";

        for(let i = 1; i <= 17; i++) {
            let colA = escalaA[i];
            let colB = escalaB[i];

            // Renderiza Linha A
            let elA = document.getElementById(ch-A-${i});
            if (elA) {
                if (colA && colA.nome) {
                    let pClass = classeProdutividade(colA.produtividade);
                    elA.innerHTML = 
                        <span style="display:flex; align-items:center; overflow:hidden; text-overflow:ellipsis;">
                            <span class="status-bolinha ${pClass}" title="Produtividade"></span>${colA.nome}
                        </span>
                        <span class="info-metricas">
                            <span title="Produtividade">P: ${colA.produtividade ?? '-'}</span>
                            <span title="Ociosidade">O: ${colA.ociosidade ?? '-'}</span>
                        </span>;
                } else {
                    elA.innerHTML = "---";
                }
            }

            // Renderiza Linha B
            let elB = document.getElementById(ch-B-${i});
            if (elB) {

    function gerarInputsConfig() {
        document.getElementById('in-nome-tl').value = nomeTL;
        const containerA = document.getElementById('inputs-linha-A');
        const containerB = document.getElementById('inputs-linha-B');

        containerA.innerHTML = '';
        containerB.innerHTML = '';

        for(let i = 1; i <= 17; i++) {
            containerA.innerHTML += 
            <div class="config-group">
                <label>Bancada ${i}</label>
                <input type="text" id="in-A-${i}" value="${escalaA[i]?.nome || ''}">
            </div>;

            containerB.innerHTML += 
            <div class="config-group">
                <label>Bancada ${i}</label>
                <input type="text" id="in-B-${i}" value="${escalaB[i]?.nome || ''}">
            </div>;
        }
    }

    function salvarConfig() {
        nomeTL = document.getElementById('in-nome-tl').value;

        for(let i = 1; i <= 17; i++) {
            let inputA = document.getElementById(in-A-${i});
            let inputB = document.getElementById(in-B-${i});

            if (inputA && inputB) {
                let nomeA = inputA.value.trim();
                let nomeB = inputB.value.trim();

                if(!escalaA[i]) escalaA[i] = {};
                escalaA[i].nome = nomeA;

                if(!escalaB[i]) escalaB[i] = {};
                escalaB[i].nome = nomeB;
            }
        }

        render();

        localStorage.setItem(
            'meli_checkin_duplo_save',
            JSON.stringify({ escalaA, escalaB, nomeTL })
        );

        toggleConfig();
        alert("Escala do Check-in salva com sucesso!");
    }

    function toggleConfig() {
        const aba = document.getElementById('aba-config');
        if(aba.style.display === 'block') {
            aba.style.display = 'none';
        } else {
            gerarInputsConfig();
            aba.style.display = 'block';
            aba.scrollIntoView({ behavior:'smooth' });
        }
    }

    function iniciarRelogio() {
        setInterval(() => {
            const agora = new Date();
            document.getElementById('relogio').innerText =
                agora.toLocaleDateString('pt-BR') + ' | ' + agora.toLocaleTimeString('pt-BR');
        }, 1000);
    }

    function sincronizarEquipeDoDia() {
    var tlLdap = document.getElementById('inputTlLdap').value.trim();
    
    if (!tlLdap) {
        alert("Por favor, digite o seu LDAP.");
        return;
    }

    const urlAppScript = "https://script.google.com/a/macros/mercadolivre.com/s/AKfycbzg-q_5NQQlAy2gUfMXoFd9FK1CPgrrkjr9OvVfMtl6yVbrA2n-SGvvw6rtzm3Lu_7Y/exec" + encodeURIComponent(tlLdap);

    alert("Buscando equipe para o LDAP: " + tlLdap + "...");

    fetch(urlAppScript)
        .then(response => response.json())
        .then(listaOperadores => {
            if (!listaOperadores || listaOperadores.length === 0) {
                alert("Nenhum operador encontrado para este LDAP na extração de hoje.");
                return;
            }

            nomeTL = tlLdap.toUpperCase();
            document.getElementById('in-nome-tl').value = nomeTL;

            escalaA = Array(18).fill(null);
            escalaB = Array(18).fill(null);

            let indexA = 1;
            let indexB = 1;

            listaOperadores.forEach((op, idx) => {
                let dadosOp = {
                    nome: op.meli_user,
                    produtividade: op.produtividade || '-',
                    ociosidade: op.ociosidade || '-'
                };

                if (idx % 2 === 0) {
                    if (indexA <= 17) {
                        escalaA[indexA] = dadosOp;
                        indexA++;
                    }
                } else {
                    if (indexB <= 17) {
                        escalaB[indexB] = dadosOp;
                        indexB++;
                    }
                }
            });

            gerarInputsConfig();
            alert("Sucesso! " + listaOperadores.length + " operadores sincronizados. Revise as posições se necessário e clique em 'Salvar Escala no Painel'.");
        })
        .catch(erro => {
            console.error("Erro ao sincronizar: ", erro);
            alert("Erro de conexão com o Apps Script. Verifique se o Web App está publicado para 'Qualquer pessoa'.");
        });
}

const URL_API_GOOGLE = "https://script.google.com/macros/s/AKfycbygAM9GtWx02JEQDKIzx7zEdVYsMkOzggq-iAzhBaZLkset4pFdvXrRCz8Tc6er-4il/exec";

function carregarDadosTempoReal() {
    fetch(URL_API_GOOGLE)
    .then(response => response.json())
    .then(registros => {
        if(registros && registros.length > 0) {
            if(registros[0].team_leader && !nomeTL) {
                nomeTL = registros[0].team_leader;
            }

            registros.forEach(item => {
                let nomeMeli = (item.meli_user || "").trim().toLowerCase();
                if(!nomeMeli) return;

                let pVal = (item.produtividade !== "" && item.produtividade !== undefined && item.produtividade !== null) ? item.produtividade : ((item.process_duration !== "" && item.process_duration !== undefined && item.process_duration !== null) ? item.process_duration : '-');
                let oVal = (item.ociosidade !== "" && item.ociosidade !== undefined && item.ociosidade !== null) ? item.ociosidade : ((item.idle_time !== "" && item.idle_time !== undefined && item.idle_time !== null) ? item.idle_time : '-');

                for(let i = 1; i <= 17; i++) {
                    if(escalaA[i] && escalaA[i].nome && escalaA[i].nome.trim().toLowerCase() === nomeMeli) {
                        escalaA[i].produtividade = pVal;
                        escalaA[i].ociosidade = oVal;
                    }
                    if(escalaB[i] && escalaB[i].nome && escalaB[i].nome.trim().toLowerCase() === nomeMeli) {
                        escalaB[i].produtividade = pVal;
                        escalaB[i].ociosidade = oVal;
                    }
                }
            });

            render();
        }
    })
